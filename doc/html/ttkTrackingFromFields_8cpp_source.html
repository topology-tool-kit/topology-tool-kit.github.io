606 newGrid->GetCellData()->AddArray(surfMeanArr);
607 }
608
609 output->ShallowCopy(newGrid);
610
611 if(DoMergeTree)
612 writeSegmentationArrays(segOutput, vertexTrajPerFrame);
613
614 this->printMsg(ttk::debug::Separator::L2);
615 return 1;
616}

618void ttkTrackingFromFields::writeSegmentationArrays(
619 vtkDataSet *segOutput,
620 const std::vector<std::vector<int>> &vertexTrajPerFrame) {
621
622 if(vertexTrajPerFrame.empty())
623 return;
624
625 const vtkIdType nPts = segOutput->GetNumberOfPoints();
626 const int nFrames = static_cast<int>(vertexTrajPerFrame.size());
627
628 for(int frame = 0; frame < nFrames; ++frame) {
629 const auto &labels = vertexTrajPerFrame[frame];
630 if(static_cast<vtkIdType>(labels.size()) != nPts) {
631 this->printWrn("Segmentation output size mismatch on frame "
632 + std::to_string(frame));
633 continue;
634 }
635 char segName[20];
636 std::snprintf(segName, sizeof(segName), "Seg_%04d", frame);
637 vtkNew<vtkIntArray> segArr;
638 segArr->SetName(segName);
639 segArr->SetNumberOfComponents(1);
640 segArr->SetNumberOfTuples(nPts);
641 for(vtkIdType v = 0; v < nPts; ++v)
642 segArr->SetValue(v, labels[v]);
643 segOutput->GetPointData()->AddArray(segArr);
644 }

647int ttkTrackingFromFields::RequestData(vtkInformation *ttkNotUsed(request),
648 vtkInformationVector **inputVector,
649 vtkInformationVector *outputVector) {
650
651 auto input = vtkDataSet::GetData(inputVector[0]);
652 auto output = vtkUnstructuredGrid::GetData(outputVector, 0);
653 auto segOutput = vtkDataSet::GetData(outputVector, 1);
654 ttk::Triangulation *triangulation = ttkAlgorithm::GetTriangulation(input);
655 if(!triangulation)
656 return 0;
657
658 this->preconditionTriangulation(triangulation);
659
660 if(input == nullptr || output == nullptr || segOutput == nullptr) {
661 return -1;
662 }
663 segOutput->ShallowCopy(input);
664 std::vector<vtkDataArray *> inputScalarFieldsRaw;
665 std::vector<vtkDataArray *> inputScalarFields;
666 const auto pointData = input->GetPointData();
667 int numberOfInputFields = pointData->GetNumberOfArrays();
668 if(numberOfInputFields < 3) {
669 this->printErr("Not enough input fields to perform tracking.");
670 }
671
672 vtkDataArray *firstScalarField = pointData->GetArray(0);
673
674 for(int i = 0; i < numberOfInputFields; ++i) {
675 vtkDataArray *currentScalarField = pointData->GetArray(i);
676 if(currentScalarField == nullptr
677 || currentScalarField->GetName() == nullptr) {
678 continue;
679 }
680 std::string const sfname{currentScalarField->GetName()};
681 if(sfname.rfind("_Order") == (sfname.size() - 6)) {
682 continue;
683 }
684 if(firstScalarField->GetDataType() != currentScalarField->GetDataType()) {
685 this->printErr("Inconsistent field data type or size between fields `"
686 + std::string{firstScalarField->GetName()} + "' and `"
687 + sfname + "'");
688 return -1;
689 }
690 inputScalarFieldsRaw.push_back(currentScalarField);
691 }
692
693 std::sort(inputScalarFieldsRaw.begin(), inputScalarFieldsRaw.end(),
694 [](vtkDataArray *a, vtkDataArray *b) {
695 std::string s1 = a->GetName();
696 std::string s2 = b->GetName();
697 return std::lexicographical_compare(
698 s1.begin(), s1.end(), s2.begin(), s2.end());
699 });
700
701 numberOfInputFields = inputScalarFieldsRaw.size();
702 int const end = EndTimestep <= 0 ? numberOfInputFields
703 : std::min(numberOfInputFields, EndTimestep);
704 for(int i = StartTimestep; i < end; i += Sampling) {
705 vtkDataArray *currentScalarField = inputScalarFieldsRaw[i];
706 // Print scalar field names:
707 // std::cout << currentScalarField->GetName() << std::endl;
708 inputScalarFields.push_back(currentScalarField);
709 }
710
711 // Input -> persistence filter.
712 std::string const algorithm = DistanceAlgorithm;
713 int const pvalg = PVAlgorithm;
714 bool useTTKMethod = false;
715 bool trackWithCriticalPoints = (pvalg == 2);
716
717 if(pvalg >= 0) {
718 switch(pvalg) {
719 case 0:
720 case 1:
721 case 2:
722 case 3:
723 useTTKMethod = true;
724 break;
725 case 4:
726 break;
727 default:
728 this->printMsg("Unrecognized tracking method.");
729 break;
730 }
731 } else {
732 using ttk::str2int;
733 switch(str2int(algorithm.c_str())) {
734 case str2int("0"):
735 case str2int("ttk"):
736 case str2int("1"):
737 case str2int("legacy"):
738 case str2int("2"):
739 case str2int("geometric"):
740 case str2int("3"):
741 case str2int("parallel"):
742 useTTKMethod = true;
743 break;
744 case str2int("4"):
745 case str2int("greedy"):
746 break;
747 default:
748 this->printMsg("Unrecognized tracking method.");
749 break;
750 }
751 }
752
753 // 0. get data
754 int const fieldNumber = inputScalarFields.size();
755 std::vector<void *> inputFields(fieldNumber);
756 for(int i = 0; i < fieldNumber; i++) {
757 inputFields[i] = ttkUtils::GetVoidPointer(inputScalarFields[i]);
758 }
759 this->setInputScalars(inputFields);
760
761 // 0'. get offsets
762 std::vector<ttk::SimplexId *> inputOrders(fieldNumber);
763 for(int i = 0; i < fieldNumber; ++i) {
764 this->SetInputArrayToProcess(0, 0, 0, 0, inputScalarFields[i]->GetName());
765 auto orderArray
766 = this->GetOrderArray(input, 0, triangulation, false, 0, false);
767 inputOrders[i]
768 = static_cast<ttk::SimplexId *>(ttkUtils::GetVoidPointer(orderArray));
769 }
770 this->setInputOffsets(inputOrders);
771
772 int status = 0;
773 this->printMsg("Tracking trajectories over " + std::to_string(fieldNumber)
774 + " timesteps");
775 if(useTTKMethod && !trackWithCriticalPoints) {
776 ttkVtkTemplateMacro(
777 inputScalarFields[0]->GetDataType(), triangulation->getType(),
778 (status = this->trackWithPersistenceMatching<VTK_TT, TTK_TT>(
779 output, fieldNumber, (TTK_TT *)triangulation->getData())));
780 } else if(useTTKMethod && trackWithCriticalPoints) {
781 ttkVtkTemplateMacro(
782 inputScalarFields[0]->GetDataType(), triangulation->getType(),
783 (status = this->trackWithCriticalPointMatching<VTK_TT, TTK_TT>(
784 output, fieldNumber, (TTK_TT *)triangulation->getData())));
785 } else {
786 this->printMsg("The specified matching method is not supported.");
787 }
788
789 if(status == 1 && EnablePostProc) {
790 ttkVtkTemplateMacro(inputScalarFields[0]->GetDataType(),
791 triangulation->getType(),
792 (this->applyPostProcessing<VTK_TT, TTK_TT>(
793 output, segOutput, input, inputScalarFields,
794 (TTK_TT *)triangulation->getData())));
795 }
796 return status;
797}
