142 template <typename dataType>
143 vectorValue getVectorValueAt(SimplexId &vertex) const {
144#ifndef TTK_ENABLE_KAMIKAZE
145 if(inputVectorField_.first == nullptr) {
146 this->printErr(
147 "Attempting to access vector before setting input vector field");
148 return {};
149 }
150#endif
151 vectorValue result;
152 const dataType *dataArray
153 = reinterpret_cast<const dataType *>(inputVectorField_.first);
154 result[0] = static_cast<float>(dataArray[3 * vertex + 0]);
155 result[1] = static_cast<float>(dataArray[3 * vertex + 1]);
156 result[2] = static_cast<float>(dataArray[3 * vertex + 2]);
157
158 return result;
159 }

164 inline void preconditionTriangulation(AbstractTriangulation *const data) {
165 if(data != nullptr) {
166 const auto dim{data->getDimensionality()};
167
168 data->preconditionBoundaryVertices();
169 data->preconditionVertexNeighbors();
170 data->preconditionVertexEdges();
171 data->preconditionVertexStars();
172 data->preconditionEdges();
173 data->preconditionEdgeStars();
174
175 data->preconditionEdgeTriangles();
176 data->preconditionTriangles();
177 if(dim >= 2) {
178 data->preconditionBoundaryEdges();
179 }
180 if(dim == 2) {
181 data->preconditionCellEdges();
182 } else if(dim == 3) {
183 data->preconditionBoundaryTriangles();
184 data->preconditionVertexTriangles();
185 data->preconditionTriangleEdges();
186 data->preconditionTriangleStars();
187 data->preconditionCellTriangles();
188 }
