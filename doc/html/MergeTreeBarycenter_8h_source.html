376 ftm::idNode getNodesAndScalarsToAdd(
377 ftm::idNode nodeId1,
378 ftm::FTMTree_MT *tree,
379 ftm::idNode nodeId2,
380 std::vector<dataType> &newScalarsVector,
381 std::vector<std::tuple<ftm::idNode, ftm::idNode, int>> &nodesToProcess,
382 ftm::idNode nodeCpt,
383 int i) {
384 // Get nodes and scalars to add
385 std::queue<std::tuple<ftm::idNode, ftm::idNode>> queue;
386 queue.emplace(nodeId2, nodeId1);
387 nodesToProcess.emplace_back(nodeId2, nodeId1, i);
388 while(!queue.empty()) {
389 auto &queueTuple = queue.front();
390 queue.pop();
391 ftm::idNode const node = std::get<0>(queueTuple);
392 // Get scalars
393 newScalarsVector.push_back(
394 tree->getValue<dataType>(tree->getNode(node)->getOrigin()));
395 newScalarsVector.push_back(tree->getValue<dataType>(node));
396 // Process children
397 std::vector<ftm::idNode> children;
398 tree->getChildren(node, children);
399 for(auto child : children) {
400 queue.emplace(child, nodeCpt + 1);
401 nodesToProcess.emplace_back(child, nodeCpt + 1, i);
402 }
403 nodeCpt += 2; // we will add two nodes (birth and death)
404 }
405
406 return nodeCpt;

462 template <class dataType>
463 void updateBarycenterTreeStructure(
464 std::vector<ftm::FTMTree_MT *> &trees,
465 ftm::MergeTree<dataType> &baryMergeTree,
466 std::vector<std::vector<std::tuple<ftm::idNode, ftm::idNode, double>>>
467 &matchings) {
468 ftm::FTMTree_MT *baryTree = &(baryMergeTree.tree);
469 ftm::idNode const baryTreeRoot = baryTree->getRoot();
470
471 // Init matching matrix
472 // m[i][j] contains the node in the barycenter matched to the jth node of
473 // the ith tree
474 std::vector<std::vector<ftm::idNode>> matrixMatchings(trees.size());
475 std::vector<bool> baryMatched(baryTree->getNumberOfNodes(), false);
476 for(unsigned int i = 0; i < matchings.size(); ++i) {
477 auto &matching = matchings[i];
478 matrixMatchings[i].resize(trees[i]->getNumberOfNodes(),
479 std::numeric_limits<ftm::idNode>::max());
480 for(auto &match : matching) {
481 matrixMatchings[i][std::get<1>(match)] = std::get<0>(match);
482 baryMatched[std::get<0>(match)] = true;
483 }
484 }
485
486 // Iterate through trees to get the nodes to add in the barycenter
487 std::vector<std::vector<ftm::idNode>> nodesToAdd(trees.size());
488#ifdef TTK_ENABLE_OPENMP4
489#pragma omp parallel for schedule(dynamic) \
490 num_threads(this->threadNumber_) if(parallelize_)
491#endif
492 for(unsigned int i = 0; i < trees.size(); ++i) {
493 ftm::idNode const root = trees[i]->getRoot();
494 std::queue<ftm::idNode> queue;
495 queue.emplace(root);
496 while(!queue.empty()) {
497 ftm::idNode const node = queue.front();
498 queue.pop();
499 bool processChildren = true;
500 // if node in trees[i] is not matched
501 if(matrixMatchings[i][node]
502 == std::numeric_limits<ftm::idNode>::max()) {
503 if(not keepSubtree_) {
504 processChildren = false;
505 nodesToAdd[i].push_back(node);
506 } else {
507 // not todo manage if keepSubtree=true (not important since it is
508 // not a valid merge tree)
509 printErr(
510 "barycenter with keepSubtree_=true is not implemented yet");
511 }
512 }
513 if(processChildren) {
514 std::vector<ftm::idNode> children;
515 trees[i]->getChildren(node, children);
516 for(auto child : children)

518 and trees[i]->isLeaf(child)))
519 queue.emplace(child);
520 }
521 }
522 }
523
524 bool foundRootNotMatched = false;
525 for(unsigned int i = 0; i < trees.size(); ++i)
526 foundRootNotMatched |= baryTree->isNodeIdInconsistent(
527 matrixMatchings[i][trees[i]->getRoot()]);
528 if(foundRootNotMatched)

531
532 // Delete nodes that are not matched in the barycenter
533 for(unsigned int i = 0; i < baryTree->getNumberOfNodes(); ++i)
534 if(not baryMatched[i])
535 baryTree->deleteNode(i);
536
537 if(not keepSubtree_) {
538 // Add scalars and nodes not present in the barycenter
539 ftm::idNode nodeCpt = baryTree->getNumberOfNodes();
540 std::vector<std::tuple<ftm::idNode, ftm::idNode, int>> nodesToProcess;
541 std::vector<dataType> newScalarsVector;
542 ftm::getTreeScalars<dataType>(baryMergeTree, newScalarsVector);
543 for(unsigned int i = 0; i < nodesToAdd.size(); ++i) {
544 for(auto node : nodesToAdd[i]) {
545 ftm::idNode parent
546 = matrixMatchings[i][trees[i]->getParentSafe(node)];
547 if(matchings[i].size() == 0)
548 parent = baryTreeRoot;

550 if((baryTree->isNodeIdInconsistent(parent)

608 template <class dataType>
609 std::tuple<dataType, dataType>
610 interpolation(ftm::MergeTree<dataType> &baryMergeTree,
611 ftm::idNode nodeId,
612 std::vector<dataType> &newScalarsVector,
613 std::vector<ftm::FTMTree_MT *> &trees,
614 std::vector<ftm::idNode> &nodes,
615 std::vector<double> &alphas) {
616 dataType newBirth = 0, newDeath = 0;
617
618 // Compute projection
619 dataType tempBirth = 0, tempDeath = 0;
620 double alphaSum = 0;
621 for(unsigned int i = 0; i < trees.size(); ++i)
622 if(nodes[i] != std::numeric_limits<ftm::idNode>::max())
623 alphaSum += alphas[i];
624 for(unsigned int i = 0; i < trees.size(); ++i) {
625 // if node is matched in trees[i]
626 if(nodes[i] != std::numeric_limits<ftm::idNode>::max()) {
627 auto iBirthDeath
628 = getParametrizedBirthDeath<dataType>(trees[i], nodes[i]);
629 dataType tTempBirth = 0, tTempDeath = 0;
630 tTempBirth += std::get<0>(iBirthDeath);
631 tTempDeath += std::get<1>(iBirthDeath);
632 tempBirth += tTempBirth * alphas[i] / alphaSum;
633 tempDeath += tTempDeath * alphas[i] / alphaSum;
634 }
635 }
636 dataType const projec = (tempBirth + tempDeath) / 2;
637
638 // Compute newBirth and newDeath
639 for(unsigned int i = 0; i < trees.size(); ++i) {
640 dataType iBirth = projec, iDeath = projec;
641 // if node is matched in trees[i]
642 if(nodes[i] != std::numeric_limits<ftm::idNode>::max()) {
643 auto iBirthDeath
644 = getParametrizedBirthDeath<dataType>(trees[i], nodes[i]);
645 iBirth = std::get<0>(iBirthDeath);
646 iDeath = std::get<1>(iBirthDeath);
647 }
648 newBirth += alphas[i] * iBirth;
649 newDeath += alphas[i] * iDeath;
650 }
651 if(normalizedWasserstein_) {
652 ftm::FTMTree_MT *baryTree = &(baryMergeTree.tree);
653 dataType mu_max = getMinMaxLocalFromVector<dataType>(
654 baryTree, nodeId, newScalarsVector, false);
655 dataType mu_min = getMinMaxLocalFromVector<dataType>(
656 baryTree, nodeId, newScalarsVector);
657 // Forbid compiler optimization to have same results on different
658 // computers
659 volatile dataType tempBirthT = newBirth * (mu_max - mu_min);
660 volatile dataType tempDeathT = newDeath * (mu_max - mu_min);
661 newBirth = tempBirthT + mu_min;
662 newDeath = tempDeathT + mu_min;
663 }
664
665 return std::make_tuple(newBirth, newDeath);
666 }

704 ftm::MergeTree<dataType> &baryMergeTree,
705 std::vector<double> &alphas,
706 unsigned int indexAddedNodes,
707 std::vector<std::vector<std::tuple<ftm::idNode, ftm::idNode, double>>>
708 &matchings) {
709 ftm::FTMTree_MT *baryTree = &(baryMergeTree.tree);
710 bool const isJT = baryTree->isJoinTree<dataType>();
711
712 // Init matching matrix
713 // m[i][j] contains the node in trees[j] matched to the node i in the
714 // barycenter
715 std::vector<std::vector<ftm::idNode>> baryMatching(
716 indexAddedNodes,
717 std::vector<ftm::idNode>(
718 trees.size(), std::numeric_limits<ftm::idNode>::max()));
719 std::vector<std::tuple<int, ftm::idNode>> nodesAddedTree(
720 baryTree->getNumberOfNodes(), std::make_tuple(-1, -1));
721 for(unsigned int i = 0; i < matchings.size(); ++i) {
722 auto &matching = matchings[i];
723 for(auto &match : matching) {
724 if(std::get<0>(match) >= indexAddedNodes)
725 // get the tree of this added node
726 nodesAddedTree[std::get<0>(match)]
727 = std::make_tuple(i, std::get<1>(match));
728 else
729 baryMatching[std::get<0>(match)][i] = std::get<1>(match);
730 }
731 }
732
733 // Interpolate scalars
734 std::vector<dataType> newScalarsVector(baryTree->getNumberOfNodes());
735 ftm::idNode const root = baryTree->getRoot();
736 std::queue<ftm::idNode> queue;
737 queue.emplace(root);
738 while(!queue.empty()) {
739 ftm::idNode const node = queue.front();
740 queue.pop();
741 std::tuple<dataType, dataType> newBirthDeath;
742 if(node < indexAddedNodes) {
743 newBirthDeath
744 = interpolation<dataType>(baryMergeTree, node, newScalarsVector,
745 trees, baryMatching[node], alphas);
746 } else {
747 int const i = std::get<0>(nodesAddedTree[node]);
748 ftm::idNode const nodeT = std::get<1>(nodesAddedTree[node]);
749 newBirthDeath = interpolationAdded<dataType>(
750 trees[i], nodeT, alphas[i], baryMergeTree, node, newScalarsVector);
751 }
752 dataType nodeScalar
753 = (isJT ? std::get<1>(newBirthDeath) : std::get<0>(newBirthDeath));
754 dataType nodeOriginScalar
755 = (isJT ? std::get<0>(newBirthDeath) : std::get<1>(newBirthDeath));
756 newScalarsVector[node] = nodeScalar;
757 newScalarsVector[baryTree->getNode(node)->getOrigin()]
758 = nodeOriginScalar;
759 std::vector<ftm::idNode> children;
760 baryTree->getChildren(node, children);
761 for(auto child : children)
762 queue.emplace(child);
763 }
764
765 if(baryMergeTree.tree.isFullMerge()) {
766 auto mergedRootOrigin = baryTree->getMergedRootOrigin<dataType>();
767 dataType mergedRootOriginScalar = 0.0;
768 for(unsigned int i = 0; i < trees.size(); ++i)
769 mergedRootOriginScalar += trees[i]->getValue<dataType>(
770 trees[i]->getMergedRootOrigin<dataType>());
771 mergedRootOriginScalar /= trees.size();
772 newScalarsVector[mergedRootOrigin] = mergedRootOriginScalar;
773 }
774
775 setTreeScalars(baryMergeTree, newScalarsVector);
776 std::vector<ftm::idNode> deletedNodesT;
777 persistenceThresholding<dataType>(
778 &(baryMergeTree.tree), 0, deletedNodesT);
779 limitSizeBarycenter(baryMergeTree, trees);
780 ftm::cleanMergeTree<dataType>(baryMergeTree);
781 }

783 template <class dataType>
784 void updateBarycenterTree_path(
785 std::vector<ftm::FTMTree_MT *> &trees,
786 ftm::MergeTree<dataType> &baryMergeTree,
787 std::vector<double> &alphas,
788 std::vector<std::vector<std::pair<std::pair<ftm::idNode, ftm::idNode>,
789 std::pair<ftm::idNode, ftm::idNode>>>>
790 &matchings) {
791 ftm::FTMTree_MT *baryTree = &(baryMergeTree.tree);
792 double alphaSum = 0;
793 for(unsigned int i = 0; i < trees.size(); ++i)
794 alphaSum += alphas[i];
795 bool joinTrees = trees[0]->isJoinTree<dataType>();
796 int oldSize = baryTree->getNumberOfNodes();
797
798 // compute matched and unmatched nodes for all trees and barycenter
799 std::vector<bool> baryNodesMatched(baryTree->getNumberOfNodes(), false);
800 std::vector<std::vector<bool>> treeNodesMatched(trees.size());
801 for(unsigned int i = 0; i < trees.size(); i++) {
802 if(alphas[i] == 0)
803 continue;
804 treeNodesMatched[i].resize(trees[i]->getNumberOfNodes(), false);
805 for(auto match : matchings[i]) {
806 baryNodesMatched[match.first.first] = true;
807 baryNodesMatched[match.first.second] = true;
808 treeNodesMatched[i][match.second.first] = true;
809 treeNodesMatched[i][match.second.second] = true;
810 }
811 }
812 // compute size of new barycenter tree
813 int newSize = oldSize;
814 for(unsigned int i = 0; i < treeNodesMatched.size(); i++) {
815 if(alphas[i] == 0 || useMedianBarycenter_)
816 continue;
817 for(unsigned int j = 0; j < treeNodesMatched[i].size(); j++) {
818 if(!treeNodesMatched[i][j])
819 newSize++;
820 }
821 }
822
823 // Create new barycenter tree
824 ftm::MergeTree<dataType> baryMergeTreeNew
825 = ftm::createEmptyMergeTree<dataType>(newSize);
826 // newScalars.resize(newSize);
827 // ftm::setTreeScalars<dataType>(baryMergeTreeNew, newScalars);
828 ftm::FTMTree_MT *baryTreeNew = &(baryMergeTreeNew.tree);
829
830 // Copy the old tree structure
831 baryTreeNew->copyMergeTreeStructure(baryTree);
832
833 // delete not-matched nodes in barycenter
834 for(ftm::idNode i = 0; i < baryTree->getNumberOfNodes(); i++) {
835 if(not baryNodesMatched[i]) {
836 baryTreeNew->getNode(i)->setOrigin(-1);
837 baryTreeNew->deleteNode(i);
838 }
839 }
840
841 // relabel paths
842 std::vector<std::vector<dataType>> parentEdgeLengths(
843 baryTree->getNumberOfNodes());
844 for(unsigned int i = 0; i < trees.size(); i++) {
845 if(alphas[i] == 0)
846 continue;
847 auto tree = trees[i];
848 for(auto match : matchings[i]) {
849 dataType bv1 = baryTree->getValue<dataType>(match.first.first);
850 dataType bv2 = baryTree->getValue<dataType>(match.first.second);
851 dataType tv1 = tree->getValue<dataType>(match.second.first);
852 dataType tv2 = tree->getValue<dataType>(match.second.second);
853 dataType pathRangeB = bv1 > bv2 ? bv1 - bv2 : bv2 - bv1;
854 dataType pathRangeT = tv1 > tv2 ? tv1 - tv2 : tv2 - tv1;
855 ftm::idNode currB = baryTreeNew->getParentSafe(match.first.first);
856 ftm::idNode lastB = match.first.first;
857 while(lastB != match.first.second) {
858 dataType currValueB = baryTree->getValue<dataType>(currB);
859 dataType lastValueB = baryTree->getValue<dataType>(lastB);
860 dataType relativeValueB = lastValueB > currValueB
861 ? lastValueB - currValueB
862 : currValueB - lastValueB;
863 relativeValueB = relativeValueB / pathRangeB;
864 if(useMedianBarycenter_)
865 parentEdgeLengths[lastB].emplace_back(relativeValueB
866 * pathRangeT);
867 else
868 parentEdgeLengths[lastB].emplace_back(relativeValueB * pathRangeT
869 * alphas[i]);
870 // continue iteration
871 lastB = currB;
872 currB = baryTreeNew->getParentSafe(currB);
873 }
874 }
875 }
876 std::queue<ftm::idNode> q;
877 q.push(baryTreeNew->getRoot());
878 // std::vector<dataType> newScalars(baryTree->getNumberOfNodes(),0);
879 std::vector<dataType> newScalars(newSize, 0);
880 newScalars[baryTreeNew->getRoot()]
881 = baryTree->getValue<dataType>(baryTree->getRoot());
882 while(!q.empty()) {
883 auto curr = q.front();
884 q.pop();
885 std::vector<ftm::idNode> children;
886 baryTreeNew->getChildren(curr, children);
887 for(auto child : children) {
888 q.emplace(child);
889 if(useMedianBarycenter_) {
890 auto m = parentEdgeLengths[child].begin()
891 + parentEdgeLengths[child].size() / 2;
892 std::nth_element(parentEdgeLengths[child].begin(), m,
893 parentEdgeLengths[child].end());
894 auto medianEdgeLength
895 = parentEdgeLengths[child][parentEdgeLengths[child].size() / 2];
896 newScalars[child]
897 = newScalars[curr]
898 + (joinTrees ? -medianEdgeLength : medianEdgeLength);
899 } else {
900 dataType avgEdgeLength = 0;
901 for(auto l : parentEdgeLengths[child]) {
902 avgEdgeLength += l;
903 }
904 // avgEdgeLength =
905 // avgEdgeLength/static_cast<dataType>(trees.size());
906 avgEdgeLength = avgEdgeLength / alphaSum;
907 newScalars[child]
908 = newScalars[curr] + (joinTrees ? -avgEdgeLength : avgEdgeLength);
909 }
910 }
911 }
912 setTreeScalars(baryMergeTreeNew, newScalars);
913
914 // insert new nodes
915 int currSize = oldSize;
916 for(unsigned int i = 0; i < trees.size(); i++) {
917 if(alphas[i] == 0 || useMedianBarycenter_)
918 continue;
919 auto tree = trees[i];
920 std::vector<int> newIndices(tree->getNumberOfNodes(), -1);
921 for(auto match : matchings[i]) {
922 dataType bv1 = baryTreeNew->getValue<dataType>(match.first.first);
923 dataType bv2 = baryTreeNew->getValue<dataType>(match.first.second);
924 dataType tv1 = tree->getValue<dataType>(match.second.first);
925 dataType tv2 = tree->getValue<dataType>(match.second.second);
926 dataType pathRangeB = bv1 > bv2 ? bv1 - bv2 : bv2 - bv1;
927 dataType pathRangeT = tv1 > tv2 ? tv1 - tv2 : tv2 - tv1;
928 ftm::idNode currB = baryTreeNew->getParentSafe(match.first.first);
929 ftm::idNode currT = tree->getParentSafe(match.second.first);
930 ftm::idNode lastB = match.first.first;
931 ftm::idNode lastT = match.second.first;
932 ftm::idNode lastNode = lastB;
933 while(currB != match.first.second || currT != match.second.second) {
934 dataType currValueB = baryTreeNew->getValue<dataType>(currB);
935 dataType currValueT = tree->getValue<dataType>(currT);
936 dataType relativeValueB
937 = bv1 > bv2 ? bv1 - currValueB : currValueB - bv1;
938 dataType relativeValueT
939 = tv1 > tv2 ? tv1 - currValueT : currValueT - tv1;
940 relativeValueB = relativeValueB / pathRangeB;
941 relativeValueT = relativeValueT / pathRangeT;
942 // if next node in barycenter, ignore
943 if(relativeValueB < relativeValueT) {
944 // continue iteration
945 lastB = currB;
946 currB = baryTreeNew->getParentSafe(currB);
947 lastNode = lastB;
948 }
949 // if next node in tree, add nodes
950 else if(relativeValueB > relativeValueT) {
951 q = std::queue<ftm::idNode>();
952 std::vector<ftm::idNode> currChildren;
953 tree->getChildren(currT, currChildren);
954 newIndices[currT] = currSize; // newScalars.size();
955 currSize++;
956 ftm::idNode nI = newIndices[currT];
957 // newScalars.emplace_back(tree->getValue<dataType>(currT));
958 // newScalars.emplace_back(bv1 + (joinTrees ? relativeValueT *
959 // pathRangeB : - relativeValueT * pathRangeB));
960 newScalars[nI] = bv1
961 + (joinTrees ? relativeValueT * pathRangeB
962 : -relativeValueT * pathRangeB);
963 baryTreeNew->makeNode(nI);
964 baryTreeNew->setParent(nI, currB);
965 baryTreeNew->deleteParent(lastNode);
966 baryTreeNew->setParent(lastNode, nI);
967 baryTreeNew->getNode(nI)->setOrigin(-1);
968 std::vector<int> nodesWithoutLink;
969 // baryTreeNew->getNode(nI)->setOrigin(newIndices[tree->getNode(currT)->getOrigin()]);
970 lastNode = newIndices[currT];
971 for(auto child : currChildren) {
972 if(child == lastT)
973 continue;
974 q.emplace(child);
975 newIndices[child] = currSize; // newScalars.size();
976 currSize++;
977 nI = newIndices[child];
978 // newScalars.emplace_back(tree->getValue<dataType>(child));
979 dataType edgeLength
980 = (joinTrees ? tree->getValue<dataType>(currT)
981 - tree->getValue<dataType>(child)
982 : tree->getValue<dataType>(child)
983 - tree->getValue<dataType>(currT));
984 // newScalars.emplace_back(newScalars[newIndices[currT]] +
985 // (joinTrees ? - edgeLength * (alphas[i]/alphaSum) : edgeLength
986 // * (alphas[i]/alphaSum)));
987 newScalars[nI]
988 = newScalars[newIndices[currT]]
989 + (joinTrees ? -edgeLength * (alphas[i] / alphaSum)
990 : edgeLength * (alphas[i] / alphaSum));
991 baryTreeNew->makeNode(nI);
992 baryTreeNew->setParent(nI, newIndices[currT]);
993 baryTreeNew->getNode(nI)->setOrigin(-1);
994 if(tree->getNumberOfChildren(child) == 0
995 && newIndices[tree->getNode(child)->getOrigin()] >= 0) {
996 ftm::idNode ln
997 = newIndices[tree->getNode(child)->getOrigin()];
998 baryTreeNew->getNode(nI)->setOrigin(ln);
999 baryTreeNew->getNode(ln)->setOrigin(nI);
1000 } else {
1001 nodesWithoutLink.push_back(nI);
1002 }
1003 }
1004 while(!q.empty()) {
1005 auto currNode = q.front();
1006 q.pop();
1007 currChildren.clear();
1008 tree->getChildren(currNode, currChildren);
1009 for(auto child : currChildren) {
1010 q.emplace(child);
1011 newIndices[child] = currSize; // newScalars.size();
1012 currSize++;
1013 nI = newIndices[child];
1014 // newScalars.emplace_back(tree->getValue<dataType>(child));
1015 dataType edgeLength
1016 = (joinTrees ? tree->getValue<dataType>(currNode)
1017 - tree->getValue<dataType>(child)
1018 : tree->getValue<dataType>(child)
1019 - tree->getValue<dataType>(currNode));
1020 // newScalars.emplace_back(newScalars[newIndices[currNode]] +
1021 // (joinTrees ? - edgeLength * (alphas[i]/alphaSum) :
1022 // edgeLength * (alphas[i]/alphaSum)));
1023 newScalars[nI]
1024 = newScalars[newIndices[currNode]]
1025 + (joinTrees ? -edgeLength * (alphas[i] / alphaSum)
1026 : edgeLength * (alphas[i] / alphaSum));
1027 baryTreeNew->makeNode(nI);
1028 baryTreeNew->getNode(nI)->setOrigin(-1);
1029 baryTreeNew->setParent(nI, newIndices[currNode]);
1030 if(tree->getNumberOfChildren(child) == 0
1031 && newIndices[tree->getNode(child)->getOrigin()] >= 0) {
1032 ftm::idNode ln
1033 = newIndices[tree->getNode(child)->getOrigin()];
1034 baryTreeNew->getNode(nI)->setOrigin(ln);
1035 baryTreeNew->getNode(ln)->setOrigin(nI);
1036 } else {
1037 nodesWithoutLink.push_back(nI);
1038 }
1039 }
1040 }
1041 // std::cout <<
1042 // baryTreeNew->getNode(newIndices[currT])->getOrigin() << " " <<
1043 // nodesWithoutLink.size() << std::endl;
1044 if(baryTreeNew->getNode(newIndices[currT])->getOrigin() < 0) {
1045 baryTreeNew->getNode(newIndices[currT])
1046 ->setOrigin(nodesWithoutLink[0]);
1047 }
1048 for(ftm::idNode n : nodesWithoutLink) {
1049 baryTreeNew->getNode(n)->setOrigin(newIndices[currT]);
1050 }
1051 // continue iteration
1052 lastT = currT;
1053 currT = tree->getParentSafe(currT);
1054 } else {
1055 // this should not happen
1056 printErr("Impossible Matching behaviour.");
1057 lastB = currB;
1058 lastT = currT;
1059 currB = baryTreeNew->getParentSafe(currB);
1060 currT = tree->getParentSafe(currT);
1061 }
1062 }
1063 }
1064 setTreeScalars(baryMergeTreeNew, newScalars);
1065 }
1066
1067 ftm::cleanMergeTree<dataType>(baryMergeTreeNew, true);
1068 baryMergeTree = baryMergeTreeNew;
1069 }

1296 template <class dataType>
1297 unsigned int
1298 persistenceScaling(std::vector<ftm::FTMTree_MT *> &trees,
1299 std::vector<ftm::MergeTree<dataType>> &mergeTrees,
1300 std::vector<ftm::FTMTree_MT *> &oriTrees,
1301 int iterationNumber,
1302 std::vector<std::vector<ftm::idNode>> &deletedNodes) {
1303 deletedNodes.clear();
1304 deletedNodes.resize(oriTrees.size());
1305 unsigned int noTreesUnscaled = 0;
1306
1307 // Scale trees
1308 for(unsigned int i = 0; i < oriTrees.size(); ++i) {
1309 double persistenceThreshold = 50.0;
1310 if(iterationNumber != -1) {
1311 // Get number of pairs in scaled merge tree
1312 int const noPairs = mergeTrees[i].tree.getRealNumberOfNodes();
1313
1314 // Get pairs in original merge tree
1315 std::vector<std::tuple<ftm::idNode, ftm::idNode, dataType>> pairs;
1316 oriTrees[i]->getPersistencePairsFromTree<dataType>(
1317 pairs, branchDecomposition_);
1318
1319 // Compute new persistence threshold
1320 double const multiplier
1321 = (progressiveSpeedDivisor_ < 1e-6
1322 ? 1.
1323 : iterationNumber / progressiveSpeedDivisor_);
1324 int const decrement = multiplier * pairs.size() / 10;
1325 int thresholdIndex = pairs.size() - noPairs - std::max(decrement, 2);
1326 thresholdIndex = std::max(thresholdIndex, 0);
1327 const double persistence = std::get<2>(pairs[thresholdIndex]);
1328 persistenceThreshold
1329 = persistence / std::get<2>(pairs.back()) * 100.0;
1330 if(thresholdIndex == 0) {
1331 persistenceThreshold = 0.;
1332 ++noTreesUnscaled;
1333 }
1334 }
1335 if(persistenceThreshold != 0.) {
1336 ftm::MergeTree<dataType> mt
1337 = ftm::copyMergeTree<dataType>(oriTrees[i]);
1338 persistenceThresholding<dataType>(
1339 &(mt.tree), persistenceThreshold, deletedNodes[i]);
1340 if(mergeTrees.size() == 0)
1341 mergeTrees.resize(oriTrees.size());
1342 mergeTrees[i] = mt;
1343 trees[i] = &(mt.tree);
1344 } else {
1345 trees[i] = oriTrees[i];
1346 }
1347 }
1348
1349 printTreesStats(trees);
1350
1351 return noTreesUnscaled;
1352 }
